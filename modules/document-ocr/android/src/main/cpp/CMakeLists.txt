cmake_minimum_required(VERSION 3.22)
project(documentocr LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(OCR_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(PREFIX "${OCR_ROOT}/prebuilt/android-${ANDROID_ABI}")
foreach(lib tesseract leptonica)
  add_library(ocr_${lib} STATIC IMPORTED)
  set_target_properties(ocr_${lib} PROPERTIES IMPORTED_LOCATION "${PREFIX}/lib/lib${lib}.a")
endforeach()
add_library(documentocr SHARED bridge.cpp "${OCR_ROOT}/native/OcrEngine.cpp")
target_include_directories(documentocr PRIVATE "${OCR_ROOT}/native" "${PREFIX}/include")
target_link_libraries(documentocr PRIVATE ocr_tesseract ocr_leptonica jnigraphics android log dl)
target_link_options(documentocr PRIVATE "-Wl,-z,max-page-size=16384")
